function extractCallout(text, type) {
  const lines = text.split(/\r?\n/);
  const re = new RegExp("^>\\s*\\[!" + type + "\\]");
  for (let i = 0; i < lines.length; i++) {
    if (re.test(lines[i])) {
      let end = i + 1;
      while (end < lines.length && /^>/.test(lines[end])) end++;
      return { start: i, end, content: lines.slice(i + 1, end).map((l) => l.replace(/^>\s?/, "")) };
    }
  }
  return null;
}

const ESSAY_TS = /^-\s+(\d{2}:\d{2}|\d{2}-\d{2}\s+\d{2}:\d{2})\s+([\s\S]*)$/;
function essayParse(text) {
  // 空行分段；无时间戳的连续行合并为一段（兼容旧随笔）
  const paras = [];
  let buf = null;
  for (const l of String(text || "").split("\n")) {
    if (!l.trim()) { if (buf) { paras.push(buf); buf = null; } continue; }
    if (!buf) buf = [];
    buf.push(l);
  }
  if (buf) paras.push(buf);
  return paras.map((lines) => {
    const m = ESSAY_TS.exec(lines[0]);
    if (m) return { stamp: m[1], lines: [m[2], ...lines.slice(1)] };
    return { stamp: "", lines };
  }).filter((p) => p.lines.join("\n").trim());
}
function essaySerialize(paras) {
  // 倒序存回（最新的在最上）；时间戳段落首行 "- HH:mm 正文"，无戳段落原样
  const out = [];
  paras.forEach((p) => {
    const body = p.lines.join("\n").trim();
    if (p.stamp) out.push(`- ${p.stamp} ${body.split("\n")[0]}`, ...body.split("\n").slice(1));
    else out.push(...body.split("\n"));
    out.push("");
  });
  return out.join("\n").replace(/\n+$/, "");
}
function essayNowStamp() {
  // 当天日记只记 HH:mm；文件名日期 ≠ 今天（补写旧日记）时带 MM-DD
  const now = M();
  if (!dayStr || dayStr === now.format("YYYY-MM-DD")) return now.format("HH:mm");
  return now.format("MM-DD HH:mm");
}

/* ---- 中列：每日积累（好句子 / 好图片 / 好想法） ----
   条目存在 [!spark] callout 里，格式："- HH:mm 文本"（可换行、可 markdown），
   图片条目 "- HH:mm ![[附件.png]]"。按时间倒序渲染，最新在最上。 */
function sparkIsSep(l) { return /^\s*$/.test(l); }
function sparkParse(text) {
  // 返回条目数组 { lines:[...], raw:"原文（不含前导 '- ' 标记）", time:"HH:mm" }
  const items = [];
  let buf = null;
  for (const l of String(text || "").split("\n")) {
    const m = /^-\s+(\d{2}:\d{2})\s?(.*)$/.exec(l);
    if (m) { if (buf) items.push(buf); buf = { time: m[1], lines: [m[2]] }; }
    else if (buf) buf.lines.push(l.replace(/^\s{2}/, ""));
    else if (l.trim()) { buf = { time: "", lines: [l.replace(/^-\s?/, "")] }; } // 兼容手写无时间戳的条目
  }
  if (buf) items.push(buf);
  for (const it of items) {
    while (it.lines.length && sparkIsSep(it.lines[it.lines.length - 1])) it.lines.pop();
    while (it.lines.length && sparkIsSep(it.lines[0])) it.lines.shift();
    it.raw = it.lines.join("\n");
  }
  return items.filter((it) => it.raw.trim());
}
function sparkSerialize(items) {
  // 倒序存回（最新的在最上）；每个条目第一行 "- HH:mm ..."，续行两空格缩进
  const out = [];
  for (const it of items) {
    const head = it.time ? `- ${it.time}` : "-";
    it.lines.forEach((l, i) => out.push(i === 0 ? `${head} ${l}`.trimEnd() : `  ${l}`));
  }
  return out.join("\n");
}
function imageLink(md) {
  // ![[xxx.png|alt]] → {name, alt}
  const m = /^!\[\[([^\]|]+)(?:\|([^\]]*))?\]\]$/.exec(md.trim());
  if (!m) return null;
  return { name: m[1].trim(), alt: m[2] || "" };
}

function lockDeskToPreview() {
  try {
    for (const leaf of app.workspace.getLeavesOfType("markdown")) {
      const v = leaf.view;
      if (!v || !v.file) continue;
      const fm = app.metadataCache.getFileCache(v.file)?.frontmatter;
      const cls = fm?.cssclasses;
      const list = Array.isArray(cls) ? cls : cls ? String(cls).split(/\s+/) : [];
      if (!list.includes("daily-desk")) continue;
      if (typeof v.getMode === "function" && v.getMode() !== "source") continue;
      const cmEl = v.contentEl?.querySelector?.(".cm-editor");
      if (cmEl && cmEl.contains(document.activeElement)) continue;
      const vs = leaf.getViewState();
      if (vs?.state && vs.state.mode !== "preview") {
        leaf.setViewState({ ...vs, state: { ...vs.state, mode: "preview" } }, { focus: false });
      }
    }
  } catch (e) { console.warn("[daily-desk] lock preview failed", e); }
}
if (!window.__ddDeskModeHook) {

}

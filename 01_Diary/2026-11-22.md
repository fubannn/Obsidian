// ---------------- callout 内容读写 ----------------
function extractCallout(text, type) {
  const lines = text.split(/\r?\n/);
  const re = new RegExp("^>\\s*\\[!" + type + "\\]");
  for (let i = 0; i < lines.length; i++) {
    if (re.test(lines[i])) {
      let end = i + 1;
      while (end < lines.length && /^>/.test(lines[end])) end++;
      return { start: i, end, content: lines.slice(i + 1, end).map((l) => l.replace(/^>\s?/, "")) };
    }
  }
  return null;
}
const textOf = (b) => (b ? b.content.join("\n").replace(/\n+$/, "").replace(/^\n+/, "") : "");

let raw = "";
try {
  const f0 = app.vault.getAbstractFileByPath(cur.file.path);
  if (f0) raw = await app.vault.read(f0);
} catch (e) {}
const blocks = { dream: extractCallout(raw, "dream"), essay: extractCallout(raw, "essay") };

async function saveCallout(type, value) {
  const f = app.vault.getAbstractFileByPath(cur.file.path);
  if (!f) return;
  const lines = (await app.vault.read(f)).split(/\r?\n/);
  const b = extractCallout(lines.join("\n"), type);
  if (!b) return;
  const body = value.length ? value.split("\n").map((l) => (l ? "> " + l : ">")) : [">"];
  lines.splice(b.start + 1, b.end - b.start - 1, ...body);
  await app.vault.process(f, (t) => lines.join("\n"));
}

}

function writingCard(title, type, placeholder) {
  const card = grid.createDiv({ cls: "dd-card dd-write " + (type === "dream" ? "dd-dream" : "dd-essay") });
  const head = card.createDiv({ cls: "dd-head" });
  head.createDiv({ cls: "dd-title", text: title });
  head.createSpan({ cls: "dd-hint", text: "✎ 点按书写" });
  const body = card.createDiv({ cls: "dd-body" });

  let text = textOf(blocks[type]);

  function renderView() {
    body.textContent = "";
    if (!text) {
      body.createDiv({ cls: "dd-placeholder", text: placeholder });
      return;
    }
    try {
      dv.el("div", text, { container: body, cls: "dd-md" });
    } catch (e) {
      body.createDiv({ cls: "dd-md", text });
    }
  }

  function startEdit(initial) {
    if (card.classList.contains("is-editing")) return;
    card.classList.add("is-editing");
    body.textContent = "";
    const ta = document.createElement("textarea");
    ta.className = "dd-editor";
    ta.value = initial != null ? initial : text;
    ta.spellcheck = false;
    body.appendChild(ta);
    window.__ddEdit = { owner: cur.file.path, type, value: ta.value };
    setTimeout(() => ta.focus(), 0);
    ta.addEventListener("input", () => { if (window.__ddEdit) window.__ddEdit.value = ta.value; });
    ta.addEventListener("blur", async () => {
      const val = ta.value;
      const mine = window.__ddEdit && window.__ddEdit.type === type && window.__ddEdit.owner === cur.file.path;
      window.__ddEdit = null;
      card.classList.remove("is-editing");
      if (mine && val !== text) {
        try { await saveCallout(type, val); } catch (e) { console.warn("[daily] 保存失败", e); }
        text = val;
      }
      renderView();
    });
    ta.addEventListener("keydown", (ev) => {
      if (ev.key === "Escape") { ev.preventDefault(); ta.value = text; ta.blur(); }
      ev.stopPropagation(); // 输入时别让按键冒泡成 Obsidian 快捷键
    });
  }

  card.addEventListener("click", (ev) => {
    if (ev.target.closest("a") || card.classList.contains("is-editing")) return;
    startEdit();
  });

  if (editing && editing.type === type) startEdit(editing.value);
  else renderView();
}

if (!day) {
  list.createDiv({ cls: "dd-empty", text: "文件名需含 YYYY-MM-DD 才能联动日程" });
} else {
  const ds = day.startOf("day");
  const de = day.endOf("day");
  const rows = [];
  for (const p of dv.pages('"05_Calendar"')) {
    const folder = (p.file.folder || "").split("/").pop();
    const cal = CAL[folder];
    if (!cal) continue;
    let startDay = null;
    let endDay = null;
    try { startDay = p.date ? dv.date(p.date) : p.file.day; } catch (e) {}
    if (!startDay) continue;
    try { endDay = p.endDate ? dv.date(p.endDate) : startDay; } catch (e) { endDay = startDay; }
    if (startDay > de || endDay < ds) continue;
    rows.push({
      title: p.title ? String(p.title) : p.file.name,
      path: p.file.path,
      color: cal.color,
      calName: cal.name,
      time: p.startTime
        ? String(p.startTime).slice(0, 5) + (p.endTime ? " – " + String(p.endTime).slice(0, 5) : "")
        : "全天",
      done: !!p.completed && p.completed !== "false",
      sort: p.startTime ? String(p.startTime) : "00:00",
    });
  }
  rows.sort((a, b) => a.sort.localeCompare(b.sort));

  let doneCount = rows.filter((r) => r.done).length;
  updateNote(doneCount, rows.length);

  if (!rows.length) {
    list.createDiv({ cls: "dd-empty", text: "☁️ 今天没有日程" });
  } else {
    for (const r of rows) {
      const row = list.createDiv({ cls: "dd-ev" + (r.done ? " is-done" : "") });
      const check = row.createSpan({ cls: "dd-ev-check" });
      check.style.setProperty("--c", r.color);
      check.title = r.done ? "标记为未完成" : "标记为完成";
      const tt = row.createSpan({ cls: "dd-ev-title", text: r.title });
      row.createSpan({ cls: "dd-ev-time", text: r.time });
      row.createSpan({ cls: "dd-ev-cal", text: r.calName });
      tt.addEventListener("click", () => app.workspace.openLinkText(r.path, cur.file.path, false));
      check.addEventListener("click", async () => {
        const f = app.vault.getAbstractFileByPath(r.path);
        if (!f) return;
        const nowDone = !r.done;
        try {
          await app.fileManager.processFrontMatter(f, (fm) => {
            // FCR 用 ISO 时间戳表示完成时间，false 表示未完成
            fm.completed = nowDone ? M().format("YYYY-MM-DDTHH:mm:ss") : false;
          });
        } catch (e) { console.warn("[daily] 日程状态写入失败", e); return; }
        r.done = nowDone;
        row.classList.toggle("is-done", nowDone);
        doneCount += nowDone ? 1 : -1;
        updateNote(doneCount, rows.length);
      });
    }
  }
}

/* ---- 打开日记自动切到阅读视图（桌面版式只在阅读模式渲染） ----
   用户主动 Ctrl+E 编辑（焦点在编辑器里）时不弹回，同 Home 的做法 */
function lockDeskToPreview() {
  try {
    for (const leaf of app.workspace.getLeavesOfType("markdown")) {
      const v = leaf.view;
      if (!v || !v.file) continue;
      const fm = app.metadataCache.getFileCache(v.file)?.frontmatter;
      const cls = fm?.cssclasses;
      const list = Array.isArray(cls) ? cls : cls ? String(cls).split(/\s+/) : [];
      if (!list.includes("daily-desk")) continue;
      if (typeof v.getMode === "function" && v.getMode() !== "source") continue;
      const cmEl = v.contentEl?.querySelector?.(".cm-editor");
      if (cmEl && cmEl.contains(document.activeElement)) continue;
      const vs = leaf.getViewState();
      if (vs?.state && vs.state.mode !== "preview") {
        leaf.setViewState({ ...vs, state: { ...vs.state, mode: "preview" } }, { focus: false });
      }
    }
  } catch (e) { console.warn("[daily-desk] lock preview failed", e); }
}
if (!window.__ddDeskModeHook) {
  window.__ddDeskModeHook = true;
  app.workspace.on("file-open", () => setTimeout(lockDeskToPreview, 120));
}
setTimeout(lockDeskToPreview, 200);
